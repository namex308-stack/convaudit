"use client";

import { useCallback, useId, useRef, useState } from "react";
import { UploadCloud, X, FileText } from "lucide-react";
import { cn } from "@/lib/utils";
import { Button } from "@/components/ui/button";
import {
  FILE_SIZE_GUIDANCE,
  validateToolFile,
} from "@/lib/file-validation";

export type UploadedFile = {
  file: File;
  id: string;
  previewUrl?: string;
};

type FileUploaderProps = {
  /** Accept attribute for the file input, e.g. "image/*" or ".pdf" */
  accept: string;
  /** Allow multiple files */
  multiple?: boolean;
  /** Called when files are added (with the new files only) */
  onFilesAdded?: (files: File[]) => void;
  /** Currently displayed files (controlled) */
  files: UploadedFile[];
  /** Called when a file is removed */
  onFileRemove?: (id: string) => void;
  /** Hint text shown under the dropzone */
  hint?: string;
  /** Disable the input while processing */
  disabled?: boolean;
};

/**
 * Drag-and-drop file uploader used by all tool components.
 *
 * Renders a stylized dropzone with a hidden <input type="file">. Files
 * are surfaced to the parent via onFilesAdded and rendered as chips with
 * remove buttons.
 */
export function FileUploader({
  accept,
  multiple = false,
  onFilesAdded,
  files,
  onFileRemove,
  hint,
  disabled = false,
}: FileUploaderProps) {
  const inputRef = useRef<HTMLInputElement>(null);
  const validationErrorId = useId();
  const [isDragging, setIsDragging] = useState(false);
  const [validationError, setValidationError] = useState<string | null>(null);

  const addFiles = useCallback(
    (selectedFiles: File[]) => {
      const validFiles: File[] = [];
      const errors: string[] = [];
      for (const file of selectedFiles) {
        const error = validateToolFile(file, accept);
        if (error) errors.push(error);
        else validFiles.push(file);
      }
      setValidationError(errors.length ? errors.join(" ") : null);
      if (validFiles.length > 0 && onFilesAdded) {
        onFilesAdded(multiple ? validFiles : [validFiles[0]]);
      }
    },
    [accept, multiple, onFilesAdded]
  );

  const handleDrop = useCallback(
    (e: React.DragEvent<HTMLDivElement>) => {
      e.preventDefault();
      setIsDragging(false);
      if (disabled) return;
      const dropped = Array.from(e.dataTransfer.files);
      addFiles(multiple ? dropped : dropped.slice(0, 1));
    },
    [addFiles, disabled, multiple]
  );

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (disabled) return;
    const selected = e.target.files ? Array.from(e.target.files) : [];
    if (selected.length > 0) addFiles(selected);
    // Reset so the same file can be re-selected later
    if (inputRef.current) inputRef.current.value = "";
  };

  const formatSize = (bytes: number) => {
    if (bytes < 1024) return `${bytes} B`;
    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
  };

  return (
    <div className="w-full">
      <div
        role="button"
        tabIndex={0}
        aria-label={`Upload files. ${hint ?? ""} ${FILE_SIZE_GUIDANCE}`}
        aria-describedby={validationError ? validationErrorId : undefined}
        aria-disabled={disabled}
        onClick={() => !disabled && inputRef.current?.click()}
        onKeyDown={(e) => {
          if ((e.key === "Enter" || e.key === " ") && !disabled) {
            e.preventDefault();
            inputRef.current?.click();
          }
        }}
        onDragOver={(e) => {
          e.preventDefault();
          if (!disabled) setIsDragging(true);
        }}
        onDragLeave={() => setIsDragging(false)}
        onDrop={handleDrop}
        className={cn(
          "flex min-h-[180px] cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed p-6 text-center transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2",
          isDragging
            ? "border-primary bg-primary/5"
            : "border-border hover:border-primary/50 hover:bg-muted/30",
          disabled && "cursor-not-allowed opacity-60"
        )}
      >
        <UploadCloud className="h-10 w-10 text-muted-foreground" aria-hidden />
        <p className="mt-3 text-sm font-medium">
          Drop files here, or click to browse
        </p>
        {hint && <p className="mt-1 text-xs text-muted-foreground">{hint}</p>}
        <p className="mt-2 text-xs text-muted-foreground">{FILE_SIZE_GUIDANCE}</p>
      </div>
      <input
        ref={inputRef}
        type="file"
        accept={accept}
        multiple={multiple}
        onChange={handleChange}
        className="hidden"
        tabIndex={-1}
        disabled={disabled}
      />

      {validationError && (
        <p
          id={validationErrorId}
          className="mt-3 rounded-md bg-destructive/10 p-3 text-sm text-destructive"
          role="alert"
        >
          {validationError}
        </p>
      )}

      {files.length > 0 && (
        <ul className="mt-4 space-y-2">
          {files.map((f) => (
            <li
              key={f.id}
              className="flex items-center gap-3 rounded-lg border bg-card p-3 text-sm"
            >
              <span className="flex h-9 w-9 flex-none items-center justify-center rounded-md bg-muted">
                <FileText className="h-4 w-4 text-muted-foreground" />
              </span>
              <div className="min-w-0 flex-1">
                <p className="truncate font-medium">{f.file.name}</p>
                <p className="text-xs text-muted-foreground">
                  {formatSize(f.file.size)}
                </p>
              </div>
              {!disabled && onFileRemove && (
                <Button
                  type="button"
                  variant="ghost"
                  size="icon"
                  className="h-8 w-8"
                  aria-label={`Remove ${f.file.name}`}
                  onClick={() => onFileRemove(f.id)}
                >
                  <X className="h-4 w-4" />
                </Button>
              )}
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

"use client";

import { useState } from "react";
import { Loader2, Shield, Archive } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Label } from "@/components/ui/label";
import { Slider } from "@/components/ui/slider";
import { FileUploader, type UploadedFile } from "./FileUploader";
import { CancelOperationButton } from "./CancelOperationButton";
import { useOperationCancellation } from "./useOperationCancellation";
import { downloadBlob, formatBytes, shortId } from "@/lib/download";
import { throwIfCancelled } from "@/lib/file-validation";

type ResultItem = {
  name: string;
  blob: Blob;
  originalSize: number;
  compressedSize: number;
};

export function CompressImageTool() {
  const [files, setFiles] = useState<UploadedFile[]>([]);
  const [quality, setQuality] = useState(70);
  const [isProcessing, setIsProcessing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [results, setResults] = useState<ResultItem[]>([]);
  const [error, setError] = useState<string | null>(null);
  const cancellation = useOperationCancellation();

  const handleFilesAdded = (newFiles: File[]) => {
    // Browsers may report an empty MIME type for some formats (notably
    // HEIC), so fall back to extension matching to avoid silently
    // dropping valid files.
    const accepted = newFiles.filter(
      (f) =>
        f.type.startsWith("image/") ||
        /\.(jpe?g|png|webp|gif|bmp|avif|tiff?|heic|heif)$/i.test(f.name)
    );
    setFiles((prev) => [
      ...prev,
      ...accepted.map((f) => ({ file: f, id: shortId() })),
    ]);
    setResults([]);
    setError(null);
  };

  const handleRemove = (id: string) => {
    setFiles((prev) => prev.filter((f) => f.id !== id));
  };

  const handleProcessAnotherFile = () => {
    setFiles([]);
    setResults([]);
    setError(null);
    setProgress(0);
    setQuality(70);
  };

  const compressInBrowser = async (
    signal: AbortSignal
  ): Promise<ResultItem[]> => {
    const imageCompression = (await import("browser-image-compression")).default;
    const out: ResultItem[] = [];
    for (let i = 0; i < files.length; i++) {
      throwIfCancelled(signal);
      const f = files[i].file;
      const targetMb = Math.max(
        0.05,
        (f.size / (1024 * 1024)) * (quality / 100)
      );
      const compressed = await imageCompression(f, {
        maxSizeMB: targetMb,
        maxWidthOrHeight: 4096,
        useWebWorker: true,
        initialQuality: quality / 100,
        signal,
      });
      throwIfCancelled(signal);
      const baseName = f.name.replace(/\.[^.]+$/, "");
      const ext = f.name.split(".").pop()?.toLowerCase() || "jpg";
      out.push({
        name: `${baseName}-compressed.${ext}`,
        blob: compressed,
        originalSize: f.size,
        compressedSize: compressed.size,
      });
      setProgress(Math.round(((i + 1) / files.length) * 100));
    }
    return out;
  };

  const handleCompress = async () => {
    if (files.length === 0) return;
    setIsProcessing(true);
    setProgress(0);
    setError(null);
    setResults([]);
    const controller = cancellation.start();

    try {
      setResults(await compressInBrowser(controller.signal));
    } catch (err) {
      if (controller.signal.aborted) {
        setError("Compression cancelled. No partial results were saved.");
      } else {
        console.error(err);
        setError("Failed to compress one or more images. Please try again.");
      }
    } finally {
      cancellation.finish(controller);
      setIsProcessing(false);
    }
  };

  const handleDownloadAllZip = async () => {
    setError(null);
    try {
      const { default: JSZip } = await import("jszip");
      const zip = new JSZip();
      for (const result of results) zip.file(result.name, result.blob);
      const blob = await zip.generateAsync({ type: "blob" });
      downloadBlob(blob, "convaudit-compressed-images.zip");
    } catch (err) {
      console.error(err);
      setError("Failed to prepare the ZIP download. Please try again.");
    }
  };

  return (
    <div className="space-y-5">
      <div className="flex items-center gap-2 rounded-lg border bg-muted/30 p-3 text-sm text-muted-foreground">
        <Shield className="h-4 w-4 flex-none text-emerald-500" />
        Images are compressed in your browser; the image files are not sent to our server.
      </div>

      <FileUploader
        accept="image/*,.heic,.heif,.tiff,.tif,.bmp,.gif,.avif"
        multiple
        files={files}
        onFilesAdded={handleFilesAdded}
        onFileRemove={handleRemove}
        hint="Supports common image formats. Batch processing is supported."
        disabled={isProcessing}
      />

      <div className="space-y-3">
        <div className="flex items-center justify-between">
          <Label htmlFor="quality">Quality</Label>
          <span className="text-sm font-medium tabular-nums">{quality}</span>
        </div>
        <Slider
          id="quality"
          min={10}
          max={100}
          step={5}
          value={[quality]}
          onValueChange={(v) => setQuality(v[0])}
          disabled={isProcessing}
        />
        <p className="text-xs text-muted-foreground">
          Lower quality = smaller file size. Try 60-80 for photos, 80-95 for
          screenshots.
        </p>
      </div>

      {error && (
        <p className="rounded-md bg-destructive/10 p-3 text-sm text-destructive" role="alert">
          {error}
        </p>
      )}

      {isProcessing && (
        <div className="space-y-2">
          <div className="flex items-center justify-between gap-3 text-xs">
            <span className="flex items-center gap-1.5 text-muted-foreground">
              <Loader2 className="h-3 w-3 animate-spin" />
              {cancellation.isCancelling
                ? "Stopping compression..."
                : `Compressing... ${progress}%`}
            </span>
            <CancelOperationButton
              onCancel={cancellation.cancel}
              isCancelling={cancellation.isCancelling}
            />
          </div>
          <div className="h-1.5 overflow-hidden rounded-full bg-muted">
            <div
              className="h-full bg-foreground transition-all"
              style={{ width: `${progress}%` }}
            />
          </div>
        </div>
      )}

      <Button
        onClick={handleCompress}
        disabled={files.length === 0 || isProcessing}
        className="w-full"
        size="lg"
      >
        {isProcessing ? (
          <Loader2 className="mr-2 h-4 w-4 animate-spin" />
        ) : (
          <Shield className="mr-2 h-4 w-4" />
        )}
        {isProcessing
          ? "Compressing..."
          : `Compress ${files.length || ""} Image${files.length === 1 ? "" : "s"}`}
      </Button>

      {results.length > 0 && (
        <div className="space-y-3">
          <div className="flex flex-wrap items-center justify-between gap-2">
            <h3 className="text-sm font-semibold">Results</h3>
            <div className="flex flex-wrap items-center gap-2">
              <Button size="sm" variant="outline" onClick={handleProcessAnotherFile}>
                Process another file
              </Button>
              {results.length > 1 && (
                <Button
                  size="sm"
                  variant="outline"
                  onClick={handleDownloadAllZip}
                >
                  <Archive className="mr-1 h-3.5 w-3.5" /> Download all as ZIP
                </Button>
              )}
            </div>
          </div>
          <ul className="space-y-2">
            {results.map((r, i) => {
              const savedPct = Math.round(
                (1 - r.compressedSize / r.originalSize) * 100
              );
              return (
                <li
                  key={i}
                  className="flex items-center justify-between rounded-lg border bg-card p-3 text-sm"
                >
                  <div className="min-w-0">
                    <p className="truncate font-medium">{r.name}</p>
                    <p className="text-xs text-muted-foreground">
                      {formatBytes(r.originalSize)} â†’{" "}
                      <span className="font-medium text-emerald-600 dark:text-emerald-400">
                        {formatBytes(r.compressedSize)}
                      </span>{" "}
                      ({savedPct >= 0 ? `-${savedPct}%` : `+${Math.abs(savedPct)}%`})
                    </p>
                  </div>
                  <Button
                    size="sm"
                    variant="outline"
                    onClick={() => downloadBlob(r.blob, r.name)}
                  >
                    Download
                  </Button>
                </li>
              );
            })}
          </ul>
        </div>
      )}
    </div>
  );
}

"use client";

import { useRef, useState } from "react";
import { Download, Loader2 } from "lucide-react";
import { Button } from "@/components/ui/button";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { Label } from "@/components/ui/label";
import { Progress } from "@/components/ui/progress";
import { FileUploader, type UploadedFile } from "./FileUploader";
import { CancelOperationButton } from "./CancelOperationButton";
import { useOperationCancellation } from "./useOperationCancellation";
import { downloadBytes, formatBytes, shortId } from "@/lib/download";
import { throwIfCancelled } from "@/lib/file-validation";

type Quality = "low" | "medium" | "high";

const QUALITY_MAP: Record<Quality, { scale: number; jpegQuality: number; label: string }> = {
  low: { scale: 1.0, jpegQuality: 0.45, label: "Low (smallest file)" },
  medium: { scale: 1.0, jpegQuality: 0.65, label: "Medium (recommended)" },
  high: { scale: 1.5, jpegQuality: 0.82, label: "High (best quality)" },
};

/**
 * Compress PDF tool.
 *
 * Strategy: Render each page to a JPEG image at a chosen quality using
 * pdfjs-dist, then re-assemble those images into a new PDF with pdf-lib.
 *
 * This is most effective on PDFs that contain large embedded images.
 * Text-heavy PDFs may not shrink much, and the resulting PDF is no
 * longer text-searchable.
 */
export function CompressPdfTool() {
  const [file, setFile] = useState<UploadedFile | null>(null);
  const [quality, setQuality] = useState<Quality>("medium");
  const [isProcessing, setIsProcessing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [result, setResult] = useState<{ bytes: Uint8Array; size: number } | null>(null);
  const [error, setError] = useState<string | null>(null);
  const renderTaskRef = useRef<{ cancel: () => void } | null>(null);
  const cancellation = useOperationCancellation();

  const handleFilesAdded = (newFiles: File[]) => {
    if (newFiles.length === 0) return;
    const f = newFiles[0];
    if (!f.name.toLowerCase().endsWith(".pdf")) {
      setError("Please upload a PDF file.");
      return;
    }
    setError(null);
    setFile({ file: f, id: shortId() });
    setResult(null);
  };

  const handleRemove = () => {
    setFile(null);
    setResult(null);
  };

  const handleProcessAnotherFile = () => {
    setFile(null);
    setResult(null);
    setError(null);
    setProgress(0);
    setQuality("medium");
  };

  const handleCompress = async () => {
    if (!file) return;
    setIsProcessing(true);
    setError(null);
    setProgress(0);
    setResult(null);
    const controller = cancellation.start();
    try {
      throwIfCancelled(controller.signal);
      const pdfjs = await import("pdfjs-dist/build/pdf.mjs");
      pdfjs.GlobalWorkerOptions.workerSrc = new URL(
        "pdfjs-dist/build/pdf.worker.mjs",
        import.meta.url
      ).toString();

      const { PDFDocument } = await import("pdf-lib");

      const buf = await file.file.arrayBuffer();
      throwIfCancelled(controller.signal);
      const src = await pdfjs.getDocument({ data: buf }).promise;
      const out = await PDFDocument.create();
      const cfg = QUALITY_MAP[quality];

      for (let i = 1; i <= src.numPages; i++) {
        throwIfCancelled(controller.signal);
        const page = await src.getPage(i);
        const viewport = page.getViewport({ scale: cfg.scale * 1.5 });
        const canvas = document.createElement("canvas");
        canvas.width = viewport.width;
        canvas.height = viewport.height;
        const ctx = canvas.getContext("2d");
        if (!ctx) throw new Error("Canvas not supported");
        // White background so transparent areas don't go black in JPEG
        ctx.fillStyle = "#ffffff";
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        const renderTask = page.render({ canvasContext: ctx, viewport, canvas });
        renderTaskRef.current = renderTask;
        try {
          await renderTask.promise;
        } finally {
          if (renderTaskRef.current === renderTask) renderTaskRef.current = null;
        }
        throwIfCancelled(controller.signal);
        const blob = await new Promise<Blob>((resolve, reject) =>
          canvas.toBlob(
            (b) => (b ? resolve(b) : reject(new Error("Failed to encode page"))),
            "image/jpeg",
            cfg.jpegQuality
          )
        );
        const ab = await blob.arrayBuffer();
        const jpg = await out.embedJpg(new Uint8Array(ab));
        // Preserve original page size in PDF points (1 px @ 72dpi = 1pt)
        const ptW = viewport.width / 1.5;
        const ptH = viewport.height / 1.5;
        const newPage = out.addPage([ptW, ptH]);
        newPage.drawImage(jpg, { x: 0, y: 0, width: ptW, height: ptH });
        setProgress(Math.round((i / src.numPages) * 100));
      }

      throwIfCancelled(controller.signal);
      const bytes = await out.save();
      setResult({ bytes, size: bytes.byteLength });
    } catch (err) {
      if (controller.signal.aborted) {
        setError("Compression cancelled. No partial result was saved.");
      } else {
        console.error(err);
        setError("Failed to compress PDF. The file may be corrupted or encrypted.");
      }
    } finally {
      renderTaskRef.current = null;
      cancellation.finish(controller);
      setIsProcessing(false);
    }
  };

  return (
    <div className="space-y-5">
      <FileUploader
        accept="application/pdf,.pdf"
        files={file ? [file] : []}
        onFilesAdded={handleFilesAdded}
        onFileRemove={handleRemove}
        hint="Upload a single PDF. Best results on image-heavy PDFs."
        disabled={isProcessing}
      />

      <div className="space-y-2">
        <Label htmlFor="quality-select">Compression quality</Label>
        <Select value={quality} onValueChange={(v) => setQuality(v as Quality)} disabled={isProcessing}>
          <SelectTrigger id="quality-select">
            <SelectValue />
          </SelectTrigger>
          <SelectContent>
            {(Object.keys(QUALITY_MAP) as Quality[]).map((q) => (
              <SelectItem key={q} value={q}>
                {QUALITY_MAP[q].label}
              </SelectItem>
            ))}
          </SelectContent>
        </Select>
        <p className="text-xs text-muted-foreground">
          Note: After compression, text in the PDF is no longer searchable.
        </p>
      </div>

      {error && (
        <p className="rounded-md bg-destructive/10 p-3 text-sm text-destructive" role="alert">
          {error}
        </p>
      )}

      {isProcessing && (
        <div className="space-y-2">
          <div className="flex items-center justify-between gap-3">
            <p className="text-xs text-muted-foreground">
              {cancellation.isCancelling ? "Cancelling..." : `Rendering pages... ${progress}%`}
            </p>
            <CancelOperationButton
              onCancel={() => {
                cancellation.cancel();
                renderTaskRef.current?.cancel();
              }}
              isCancelling={cancellation.isCancelling}
            />
          </div>
          <Progress value={progress} />
        </div>
      )}

      <Button
        onClick={handleCompress}
        disabled={!file || isProcessing}
        className="w-full"
        size="lg"
      >
        {isProcessing ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null}
        {isProcessing ? "Compressing..." : "Compress PDF"}
      </Button>

      {result && file && (
        <div className="rounded-lg border bg-card p-4 text-sm">
          <p className="font-medium">Compression complete!</p>
          <p className="mt-1 text-xs text-muted-foreground">
            Original: {formatBytes(file.file.size)} â†’{" "}
            <span className="font-medium text-emerald-600 dark:text-emerald-400">
              {formatBytes(result.size)}
            </span>{" "}
            ({Math.max(0, Math.round((1 - result.size / file.file.size) * 100))}% smaller)
          </p>
          <div className="mt-3 flex flex-wrap gap-2">
            <Button
              size="sm"
              onClick={() => {
                const baseName = file.file.name.replace(/\.pdf$/i, "");
                downloadBytes(result.bytes, `${baseName}-compressed.pdf`, "application/pdf");
              }}
            >
              <Download className="mr-1 h-3.5 w-3.5" /> Download compressed PDF
            </Button>
            <Button
              size="sm"
              variant="outline"
              onClick={handleProcessAnotherFile}
            >
              Process another file
            </Button>
          </div>
        </div>
      )}
    </div>
  );
}

"use client";

import { useState } from "react";
import { Download, Loader2, Archive } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Label } from "@/components/ui/label";
import { Slider } from "@/components/ui/slider";
import { Progress } from "@/components/ui/progress";
import { FileUploader, type UploadedFile } from "./FileUploader";
import { CancelOperationButton } from "./CancelOperationButton";
import { useOperationCancellation } from "./useOperationCancellation";
import { downloadBlob, formatBytes, shortId } from "@/lib/download";
import { throwIfCancelled } from "@/lib/file-validation";

type ResultItem = {
  name: string;
  blob: Blob;
  size: number;
  originalSize: number;
};

/**
 * HEIC to JPG tool.
 * Uses heic2any (which internally uses libheif.js WebAssembly) to decode
 * HEIC/HEIF images and re-encode them as JPEG blobs in the browser.
 */
export function HeicToJpgTool() {
  const [files, setFiles] = useState<UploadedFile[]>([]);
  const [quality, setQuality] = useState(85);
  const [isProcessing, setIsProcessing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [results, setResults] = useState<ResultItem[]>([]);
  const [error, setError] = useState<string | null>(null);
  const cancellation = useOperationCancellation();

  const handleFilesAdded = (newFiles: File[]) => {

  const handleFilesAdded = async (newFiles: File[]) => {
    const pdfs = newFiles.filter(
      (f) => f.type === "application/pdf" || /\.pdf$/i.test(f.name)
    );
    if (pdfs.length < newFiles.length) {
      setError("Some files were skipped â€” only PDF files are supported.");
    } else {
      setError(null);
    }
    // Read page counts for display
    const entries: PdfEntry[] = [];
    for (const f of pdfs) {
      try {
        const { PDFDocument } = await import("pdf-lib");
        const buf = await f.arrayBuffer();
        const doc = await PDFDocument.load(buf, { ignoreEncryption: true });
        entries.push({
          file: f,
          id: shortId(),
          pageCount: doc.getPageCount(),
        });
      } catch {
        entries.push({ file: f, id: shortId(), pageCount: 0 });
      }
    }
    setFiles((prev) => [...prev, ...entries]);
    setOutputUrl((prev) => {
      if (prev) URL.revokeObjectURL(prev);
      return null;

    setFiles((prev) => prev.filter((f) => f.id !== id));

    setOutputSize(null);

    setProgress(0);

    setFiles((prev) => {
      const next = [...prev];
      const target = direction === "up" ? index - 1 : index + 1;

      [next[index], next[target]] = [next[target], next[index]];
      return next;

/**
 * PDF to Word tool.
 *
 * Uses pdfjs-dist to extract text content from each page of the PDF, then
 * wraps that text in a minimal Word-compatible .doc file (HTML-based Word
 * 97-2003 format that opens in Word, Google Docs, and LibreOffice).
 *
 * Note: This focuses on accurate text extraction. Complex layouts, tables,
 * and images are NOT preserved.
 */
export function PdfToWordTool() {
  const [file, setFile] = useState<UploadedFile | null>(null);
  const [isProcessing, setIsProcessing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [extractedText, setExtractedText] = useState<string>("");
  const [clipboardStatus, setClipboardStatus] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const cancellation = useOperationCancellation();

  const handleFilesAdded = (newFiles: File[]) => {
    if (newFiles.length === 0) return;
    const f = newFiles[0];
    if (!f.name.toLowerCase().endsWith(".pdf")) {
      setError("Please upload a PDF file.");
      return;
    }
    setError(null);
    setFile({ file: f, id: shortId() });
    setExtractedText("");
    setClipboardStatus(null);
  };

  const handleRemove = () => {
    setFile(null);
    setExtractedText("");
    setClipboardStatus(null);
  };

  const handleProcessAnotherFile = () => {
    setFile(null);
    setExtractedText("");
    setError(null);
    setProgress(0);
    setClipboardStatus(null);
  };

  const handleConvert = async () => {
    if (!file) return;
    setIsProcessing(true);
    setError(null);
    setProgress(0);
    setExtractedText("");
    setClipboardStatus(null);
    const controller = cancellation.start();

    try {
      throwIfCancelled(controller.signal);
      // Lazy-load pdfjs-dist so the heavy WASM only loads when needed.
      const pdfjs = await import("pdfjs-dist/build/pdf.mjs");
      // Use a worker URL via webpack's new URL() support (Next.js compatible).
      pdfjs.GlobalWorkerOptions.workerSrc = new URL(
        "pdfjs-dist/build/pdf.worker.mjs",
        import.meta.url
      ).toString();

      const buf = await file.file.arrayBuffer();
      throwIfCancelled(controller.signal);
      const doc = await pdfjs.getDocument({ data: buf }).promise;
      const lines: string[] = [];
      for (let i = 1; i <= doc.numPages; i++) {
        throwIfCancelled(controller.signal);
        const page = await doc.getPage(i);
        const content = await page.getTextContent();
        throwIfCancelled(controller.signal);
        const text = content.items
          .map((item) => ("str" in item ? (item as { str: string }).str : ""))
          .join(" ");
        lines.push(text);
        setProgress(Math.round((i / doc.numPages) * 100));
      }
      throwIfCancelled(controller.signal);
      const fullText = lines.join("\n\n");
      setExtractedText(fullText);
    } catch (err) {
      if (controller.signal.aborted) {
        setError("Conversion cancelled. No partial document was created.");
      } else {
        console.error(err);
        setError(
          "Failed to extract text. The PDF may be scanned (images only) or corrupted."
        );
      }
    } finally {
      cancellation.finish(controller);
      setIsProcessing(false);
    }
  };

  const handleDownload = () => {
    if (!extractedText || !file) return;
    const baseName = file.file.name.replace(/\.pdf$/i, "");
    // Build a minimal Word-compatible .doc file (HTML format that Word reads).
    const escaped = extractedText
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;");

      .map((p) => `<p>${p.replace(/\n/g, "<br/>")}</p>`)
      .join("");

    const blob = new Blob([html], { type: "application/msword" });
    downloadBlob(blob, `${baseName}.doc`);
  };

      copied
        ? "Text copied to clipboard."
        : "Could not copy text. Select it from the preview and copy manually.",
    );
  };

  const handleDownloadText = () => {
    if (!extractedText || !file) return;
    const baseName = file.file.name.replace(/\.pdf$/i, "");
    const blob = new Blob([extractedText], { type: "text/plain;charset=utf-8" });
    downloadBlob(blob, `${baseName}.txt`);
  };

  return (
    <div className="space-y-5">
      <FileUploader
        accept="application/pdf,.pdf"
        files={file ? [file] : []}
        onFilesAdded={handleFilesAdded}
        onFileRemove={handleRemove}
        hint="Upload a PDF with embedded text. Scanned PDFs (images only) are not supported."
        disabled={isProcessing}

              cancellingLabel="Stopping after this page..."
            />
          </div>
          <Progress value={progress} />

      <Button
        onClick={handleConvert}
        disabled={!file || isProcessing}
        className="w-full"
        size="lg"
      >
        {isProcessing ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null}
        {isProcessing ? "Converting..." : "Convert to Word"}

                <Download className="mr-1 h-3.5 w-3.5" /> Download .txt
              </Button>
              <Button size="sm" onClick={handleDownload}>
                <Download className="mr-1 h-3.5 w-3.5" /> Download .doc
              </Button>
              <Button
                size="sm"
                variant="outline"
                onClick={handleProcessAnotherFile}
              >
                Process another file
              </Button>
            </div>
          </div>
          {clipboardStatus && (
            <p className="text-sm text-muted-foreground" role="status" aria-live="polite">
              {clipboardStatus}

          )}
          <pre className="max-h-80 overflow-y-auto rounded-lg border bg-muted/30 p-3 text-xs whitespace-pre-wrap">

            {extractedText.length > 4000 ? "\n..." : ""}

  const navigatorWithMemory = navigator as Navigator & {

  };
  const lowMemory =
    (navigatorWithMemory.deviceMemory !== undefined &&

async function prepareSourceImage(file: File): Promise<SourceImage> {
  if (typeof createImageBitmap !== "function") {
    throw new Error("This browser cannot decode images for local processing.");
  }

  const bitmap = await createImageBitmap(file);
  try {
    const originalPixels = bitmap.width * bitmap.height;
    const pixelLimit = getDevicePixelLimit();
    const scale =
      originalPixels > pixelLimit ? Math.sqrt(pixelLimit / originalPixels) : 1;
    const width = Math.max(1, Math.round(bitmap.width * scale));
    const height = Math.max(1, Math.round(bitmap.height * scale));
    const canvas = document.createElement("canvas");
    canvas.width = width;
    canvas.height = height;

      canvas.toBlob(
        (output) =>
          output
            ? resolve(output)
            : reject(new Error("Could not prepare this image for processing.")),
        "image/png",
      );
    });

    canvas.height = 0;

      width,
      height,
      ...(scale < 1
        ? {
            notice: `Downscaled from ${bitmap.width.toLocaleString()} Ã— ${bitmap.height.toLocaleString()} to ${width.toLocaleString()} Ã— ${height.toLocaleString()} pixels to fit your device's memory limit.`,
          }
        : {}),
    };
  } finally {
    bitmap.close();
  }
}

/**
 * Blur only the output's alpha channel so the transparent cutout edge is less harsh.
 */
async function featherCutout(
  output: import("@huggingface/transformers").RawImage,
): Promise<Blob> {
  if (output.channels !== 4) {
    throw new Error("The model returned an unsupported image format.");
  }

  const width = output.width;
  const height = output.height;
  const outputCanvas = document.createElement("canvas");
  const alphaCanvas = document.createElement("canvas");
  const blurredCanvas = document.createElement("canvas");
  for (const canvas of [outputCanvas, alphaCanvas, blurredCanvas]) {
    canvas.width = width;
    canvas.height = height;
  }

  const outputContext = outputCanvas.getContext("2d", {
    willReadFrequently: true,
  });
  const alphaContext = alphaCanvas.getContext("2d", {
    willReadFrequently: false,
  });
  const blurredContext = blurredCanvas.getContext("2d", {
    willReadFrequently: true,
  });
  try {
    if (!outputContext || !alphaContext || !blurredContext) {
      throw new Error("Your browser could not refine the image edges.");
    }

    const rgba = new ImageData(width, height);
    rgba.data.set(output.data);
    outputContext.putImageData(rgba, 0, 0);

    const alphaMask = new ImageData(width, height);
    for (let pixel = 0; pixel < width * height; pixel++) {
      const alpha = rgba.data[pixel * 4 + 3];
      const offset = pixel * 4;
      alphaMask.data[offset] = alpha;
      alphaMask.data[offset + 1] = alpha;
      alphaMask.data[offset + 2] = alpha;
      alphaMask.data[offset + 3] = 255;
    }
    alphaContext.putImageData(alphaMask, 0, 0);
    blurredContext.filter = "blur(1.2px)";
    blurredContext.drawImage(alphaCanvas, 0, 0);
    blurredContext.filter = "none";
    const featheredAlpha = blurredContext.getImageData(0, 0, width, height);

    for (let pixel = 0; pixel < width * height; pixel++) {
      rgba.data[pixel * 4 + 3] = featheredAlpha.data[pixel * 4];
    }
    outputContext.putImageData(rgba, 0, 0);

    return await new Promise<Blob>((resolve, reject) => {
      outputCanvas.toBlob(
        (result) =>
          result
            ? resolve(result)
            : reject(new Error("Could not create the transparent PNG.")),
        "image/png",
      );
    });
  } finally {
    for (const canvas of [outputCanvas, alphaCanvas, blurredCanvas]) {
      canvas.width = 0;
      canvas.height = 0;
    }
  }
}

function getFriendlyError(error: unknown): string {
  const message = error instanceof Error ? error.message : "";
  if (/timeout/i.test(message)) {
    return "This image took too long to process. Retry it, or use a smaller image and close other tabs.";
  }
  if (
    error instanceof RangeError ||
    /memory|allocation|ArrayBuffer|out of memory/i.test(message)
  ) {
    return "Your device ran low on memory. Close other tabs, retry with a smaller image, or process fewer images at once.";
  }
  if (/decode|unsupported|image format|corrupt/i.test(message)) {
    return "This image could not be decoded. Try a valid JPG, PNG, or WebP file.";
  }
  return "Could not remove this image's background. Try another image or retry after closing other tabs.";
}

function getOutputName(fileName: string): string {
  return `${fileName.replace(/\.[^.]+$/, "")}-no-bg.png`;
}

      const item = prev.find((f) => f.id === id);
      if (item) URL.revokeObjectURL(item.objectUrl);
      return prev.filter((f) => f.id !== id);

    setJpegQuality(95);

    item: ImageItem,
    signal: AbortSignal
  ): Promise<Blob> => {
    return new Promise((resolve, reject) => {
      const img = new Image();
      const onAbort = () => {
        img.src = "";
        reject(new DOMException("Operation cancelled.", "AbortError"));
      };
      img.onload = () => {
        signal.removeEventListener("abort", onAbort);
        try {
          throwIfCancelled(signal);
          const canvas = document.createElement("canvas");
          canvas.width = img.naturalWidth || img.width;
          canvas.height = img.naturalHeight || img.height;
          const ctx = canvas.getContext("2d");
          if (!ctx) {
            reject(new Error("Canvas context creation failed"));
            return;
          }

          // Draw directly onto blank canvas to strip all EXIF / metadata blocks
          ctx.drawImage(img, 0, 0);

          let mimeType = item.file.type;
          if (outputFormat !== "original") {
            mimeType = outputFormat;
          } else if (!mimeType || mimeType === "image/heic" || mimeType === "image/tiff") {
            mimeType = "image/jpeg";
          }

          const quality = mimeType === "image/png" ? undefined : jpegQuality / 100;

          canvas.toBlob(
            (blob) => {
              if (signal.aborted) {
                reject(new DOMException("Operation cancelled.", "AbortError"));
              } else if (blob) {
                resolve(blob);
              } else {
                reject(new Error("Failed to export cleaned image"));
              }
            },
            mimeType,
            quality
          );
        } catch (err) {
          reject(err);
        }
      };
      img.onerror = () => {
        signal.removeEventListener("abort", onAbort);
        reject(new Error("Failed to load image into memory"));
      };
      signal.addEventListener("abort", onAbort, { once: true });
      img.src = item.objectUrl;
    });

            : it

      );

      cancellation.finish(controller);
      setActiveCleanId(null);
    }

type PdfThumbnail = {
  pageNumber: number;
  dataUrl: string;
};

type RotationAngle = 90 | 180 | 270;

function getPdfErrorMessage(error: unknown): string {
  const message = error instanceof Error ? error.message : "";
  const name = error instanceof Error ? error.name : "";

  if (name === "PasswordException" || /password|encrypt/i.test(message)) {
    return "This PDF is password-protected. Remove its password before using this tool.";
  }
  if (/memory|allocation|out of memory/i.test(message)) {
    return "Your browser ran out of memory while processing this PDF. Try a smaller file or close other tabs.";
  }
  return "Could not process this PDF. The file may be corrupted or use an unsupported feature.";
}

export function RotatePdfTool() {
  const [file, setFile] = useState<UploadedFile | null>(null);
  const [thumbnails, setThumbnails] = useState<PdfThumbnail[]>([]);
  const [selectedPages, setSelectedPages] = useState<number[]>([]);
  const [rotation, setRotation] = useState<RotationAngle>(90);
  const [isLoadingPages, setIsLoadingPages] = useState(false);
  const [isProcessing, setIsProcessing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [result, setResult] = useState<Uint8Array | null>(null);
  const [error, setError] = useState<string | null>(null);

  const handleFilesAdded = async (newFiles: File[]) => {
    const selectedFile = newFiles[0];
    if (!selectedFile) return;
    if (!selectedFile.name.toLowerCase().endsWith(".pdf")) {
      setError("Please upload a PDF file.");
      return;
    }

    setFile({ file: selectedFile, id: shortId() });
    setThumbnails([]);
    setSelectedPages([]);
    setResult(null);
    setError(null);
    setProgress(0);
    setIsLoadingPages(true);

    let document: Awaited<
      ReturnType<Awaited<ReturnType<typeof import("pdfjs-dist/build/pdf.mjs")>["getDocument"]>["promise"]>
    > | null = null;
    try {
      const pdfjs = await import("pdfjs-dist/build/pdf.mjs");
      pdfjs.GlobalWorkerOptions.workerSrc = new URL(
        "pdfjs-dist/build/pdf.worker.mjs",
        import.meta.url,
      ).toString();

      const bytes = await selectedFile.arrayBuffer();
      document = await pdfjs.getDocument({ data: bytes }).promise;
      const pages: PdfThumbnail[] = [];

      for (let pageNumber = 1; pageNumber <= document.numPages; pageNumber++) {
        const page = await document.getPage(pageNumber);
        const baseViewport = page.getViewport({ scale: 1 });
        const viewport = page.getViewport({
          scale: Math.min(180 / baseViewport.width, 180 / baseViewport.height),
        });
        const canvas = window.document.createElement("canvas");
        const context = canvas.getContext("2d");
        if (!context) throw new Error("Unable to create a PDF preview canvas.");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);
        await page.render({ canvas, canvasContext: context, viewport }).promise;
        pages.push({
          pageNumber,
          dataUrl: canvas.toDataURL("image/jpeg", 0.7),
        });
        canvas.width = 0;
        canvas.height = 0;
        page.cleanup();
        setProgress(Math.round((pageNumber / document.numPages) * 100));
      }

      setThumbnails(pages);
      setSelectedPages(pages.map((page) => page.pageNumber));
    } catch (loadError) {
      console.error(loadError);
      setError(getPdfErrorMessage(loadError));
      setFile(null);
    } finally {
      await document?.destroy();
      setIsLoadingPages(false);
    }
  };

  const handleRemoveFile = () => {
    setFile(null);
    setThumbnails([]);
    setSelectedPages([]);
    setResult(null);
    setProgress(0);
    setError(null);
  };

  const togglePage = (pageNumber: number) => {
    setSelectedPages((current) =>
      current.includes(pageNumber)
        ? current.filter((page) => page !== pageNumber)
        : [...current, pageNumber].sort((a, b) => a - b),
    );
    setResult(null);
  };

  const selectAllPages = () => {
    if (selectedPages.length === thumbnails.length) {
      setSelectedPages([]);
    } else {
      setSelectedPages(thumbnails.map((page) => page.pageNumber));
    }
    setResult(null);
  };

  const handleRotate = async () => {
    if (!file || selectedPages.length === 0) return;

    setIsProcessing(true);
    setProgress(0);
    setError(null);
    setResult(null);

    try {
      const { PDFDocument, degrees } = await import("pdf-lib");
      const pdf = await PDFDocument.load(await file.file.arrayBuffer());
      const selected = new Set(selectedPages);

      for (let index = 0; index < pdf.getPageCount(); index++) {
        if (selected.has(index + 1)) {
          const page = pdf.getPage(index);
          page.setRotation(
            degrees((page.getRotation().angle + rotation) % 360),
          );
        }
        setProgress(Math.round(((index + 1) / pdf.getPageCount()) * 100));
        if (index % 10 === 9) {
          await new Promise<void>((resolve) => setTimeout(resolve, 0));
        }
      }

      setResult(await pdf.save());
      setProgress(100);
    } catch (processingError) {
      console.error(processingError);
      setError(getPdfErrorMessage(processingError));
    } finally {
      setIsProcessing(false);
    }
  };

  const isBusy = isLoadingPages || isProcessing;
  const allPagesSelected =
    thumbnails.length > 0 && selectedPages.length === thumbnails.length;
  const outputName = file
    ? `${file.file.name.replace(/\.pdf$/i, "")}-rotated.pdf`
    : "rotated.pdf";

  return (
    <div className="space-y-5">
      <FileUploader
        accept="application/pdf,.pdf"
        files={file ? [file] : []}
        onFilesAdded={handleFilesAdded}
        onFileRemove={handleRemoveFile}
        hint="Upload a PDF to preview and select pages."
        disabled={isBusy}
      />

      {thumbnails.length > 0 && (
        <>
          <div className="flex flex-wrap items-center justify-between gap-3">
            <p className="text-sm text-muted-foreground">
              {selectedPages.length} of {thumbnails.length} pages selected
            </p>
            <Button
              type="button"
              variant="outline"
              size="sm"
              onClick={selectAllPages}
              disabled={isBusy}
            >
              {allPagesSelected ? "Deselect all" : "Select all"}
            </Button>
          </div>

          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 md:grid-cols-4">
            {thumbnails.map(({ pageNumber, dataUrl }) => {
              const isSelected = selectedPages.includes(pageNumber);
              return (
                <button
                  key={pageNumber}
                  type="button"
                  aria-pressed={isSelected}
                  aria-label={`Page ${pageNumber}${isSelected ? ", selected" : ""}`}
                  onClick={() => togglePage(pageNumber)}
                  disabled={isBusy}
                  className={`rounded-lg border p-2 text-left transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring ${
                    isSelected
                      ? "border-primary bg-primary/5"
                      : "border-border hover:border-primary/50"
                  }`}
                >
                  <img
                    src={dataUrl}
                    alt={`Preview of page ${pageNumber}`}
                    className="mx-auto h-36 max-w-full object-contain"
                  />
                  <span className="mt-2 block text-center text-xs font-medium">
                    Page {pageNumber}
                  </span>
                </button>
              );
            })}
          </div>
        </>
      )}

      {thumbnails.length > 0 && (
        <div className="space-y-2">
          <label htmlFor="pdf-rotation" className="text-sm font-medium">
            Rotate clockwise
          </label>
          <select
            id="pdf-rotation"
            value={rotation}
            onChange={(event) =>
              setRotation(Number(event.target.value) as RotationAngle)
            }
            disabled={isBusy}
            className="flex h-10 w-full rounded-md border border-input bg-background px-3 py-2 text-sm ring-offset-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
          >
            <option value={90}>90 degrees</option>
            <option value={180}>180 degrees</option>
            <option value={270}>270 degrees</option>
          </select>
        </div>
      )}

      {isBusy && (
        <div className="space-y-2" role="status" aria-live="polite">
          <p className="text-xs text-muted-foreground">
            {isLoadingPages
              ? `Preparing page previews... ${progress}%`
              : `Rotating PDF... ${progress}%`}
          </p>
          <Progress value={progress} />
        </div>
      )}

      {error && (
        <p
          className="rounded-md bg-destructive/10 p-3 text-sm text-destructive"
          role="alert"
        >
          {error}
        </p>
      )}

      {thumbnails.length > 0 && (
        <Button
          type="button"
          onClick={handleRotate}
          disabled={!file || selectedPages.length === 0 || isBusy}
          className="w-full"
          size="lg"
        >
          {isProcessing ? (
            <Loader2 className="mr-2 h-4 w-4 animate-spin" />
          ) : (
            <RotateCw className="mr-2 h-4 w-4" />
          )}
          {isProcessing ? "Rotating..." : "Rotate PDF"}
        </Button>
      )}

      {result && (
        <div className="flex flex-wrap items-center justify-between gap-3 rounded-lg border border-border/70 bg-muted/30 p-4">
          <p className="min-w-0 truncate text-sm font-medium">{outputName}</p>
          <Button
            type="button"
            onClick={() =>
              downloadBytes(result, outputName, "application/pdf")
            }
          >
            <Download className="mr-2 h-4 w-4" />
            Download PDF
          </Button>
        </div>
      )}
    </div>
  );
}

"use client";

import { useState } from "react";
import { Download, Loader2, Archive } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Progress } from "@/components/ui/progress";
import { FileUploader, type UploadedFile } from "./FileUploader";
import { CancelOperationButton } from "./CancelOperationButton";
import { useOperationCancellation } from "./useOperationCancellation";
import { downloadBlob, downloadBytes, formatBytes, shortId } from "@/lib/download";
import { throwIfCancelled } from "@/lib/file-validation";

type Range = { from: number; to: number };
type SplitResult = { name: string; bytes: Uint8Array };

/**
 * Split PDF tool.
 * Parses user-entered page ranges (e.g. "1-3, 5, 7-9") and produces a
 * separate PDF for each range using pdf-lib.
 */
export function SplitPdfTool() {
  const [file, setFile] = useState<UploadedFile | null>(null);
  const [rangesInput, setRangesInput] = useState("1-1");
  const [isProcessing, setIsProcessing] = useState(false);
  const [results, setResults] = useState<SplitResult[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [pageCount, setPageCount] = useState(0);
  const [progress, setProgress] = useState(0);
  const cancellation = useOperationCancellation();

  const handleFilesAdded = async (newFiles: File[]) => {
    if (newFiles.length === 0) return;
    const f = newFiles[0];
    if (!f.name.toLowerCase().endsWith(".pdf")) {
      setError("Please upload a PDF file.");
      return;
    }
    setError(null);
    setFile({ file: f, id: shortId() });
    setResults([]);
    try {
      const { PDFDocument } = await import("pdf-lib");
      const buf = await f.arrayBuffer();
      const doc = await PDFDocument.load(buf, { ignoreEncryption: true });
      setPageCount(doc.getPageCount());
    } catch {
      setPageCount(0);
    }
  };

  const handleRemove = () => {
    setFile(null);
    setResults([]);
    setPageCount(0);
  };

  const handleProcessAnotherFile = () => {
    handleRemove();
    setError(null);
    setProgress(0);
    setRangesInput("1-1");
  };

  /**
   * Parse a string like "1-3, 5, 7-9" into an array of {from, to} ranges.
   * Whitespace is ignored. Invalid tokens are reported via thrown Error.
   */
  function parseRanges(input: string): Range[] {
    const tokens = input.split(",").map((s) => s.trim()).filter(Boolean);
    if (tokens.length === 0) throw new Error("Please enter at least one page range.");
    const ranges: Range[] = [];
    for (const token of tokens) {
      const m = /^(\d+)\s*-\s*(\d+)$|^(\d+)$/.exec(token);
      if (!m) throw new Error(`Invalid range: "${token}"`);
      if (m[1] && m[2]) {
        const from = parseInt(m[1], 10);
        const to = parseInt(m[2], 10);
        if (from > to) throw new Error(`Range start is greater than end: "${token}"`);
        ranges.push({ from, to });
      } else if (m[3]) {
        const p = parseInt(m[3], 10);
        ranges.push({ from: p, to: p });
      }
    }
    return ranges;
  }

  const handleSplit = async () => {
    if (!file) return;
    setIsProcessing(true);
    setError(null);
    setResults([]);
    setProgress(0);
    const controller = cancellation.start();
    try {
      throwIfCancelled(controller.signal);
      const ranges = parseRanges(rangesInput);
      const { PDFDocument } = await import("pdf-lib");
      const buf = await file.file.arrayBuffer();
      throwIfCancelled(controller.signal);
      const src = await PDFDocument.load(buf, { ignoreEncryption: true });
      const total = src.getPageCount();
      for (const r of ranges) {
        if (r.from < 1 || r.to > total) {
          throw new Error(
            `Range ${r.from}-${r.to} is out of bounds (PDF has ${total} pages).`
          );
        }
      }
      const out: SplitResult[] = [];
      const baseName = file.file.name.replace(/\.pdf$/i, "");
      for (let i = 0; i < ranges.length; i++) {
        throwIfCancelled(controller.signal);
        const r = ranges[i];
        const newDoc = await PDFDocument.create();
        const pageIndices: number[] = [];
        for (let p = r.from; p <= r.to; p++) pageIndices.push(p - 1);
        const pages = await newDoc.copyPages(src, pageIndices);
        throwIfCancelled(controller.signal);
        pages.forEach((p) => newDoc.addPage(p));
        const bytes = await newDoc.save();
        throwIfCancelled(controller.signal);
        const label =
          r.from === r.to ? `page-${r.from}` : `pages-${r.from}-${r.to}`;
        out.push({ name: `${baseName}-${label}.pdf`, bytes });
        setProgress(Math.round(((i + 1) / ranges.length) * 100));
      }
      setResults(out);
    } catch (err) {
      if (controller.signal.aborted) {
        setError("Split cancelled. No partial results were saved.");
      } else {
        console.error(err);
        setError(err instanceof Error ? err.message : "Failed to split PDF.");
      }
    } finally {
      cancellation.finish(controller);
      setIsProcessing(false);
    }
  };

  const handleDownloadAllZip = async () => {
    setError(null);
    try {
      const { default: JSZip } = await import("jszip");
      const zip = new JSZip();
      for (const result of results) zip.file(result.name, result.bytes);
      const blob = await zip.generateAsync({ type: "blob" });
      const base = file?.file.name.replace(/\.pdf$/i, "") ?? "split";
      downloadBlob(blob, `${base}-split.zip`);
    } catch (err) {
      console.error(err);
      setError("Failed to prepare the ZIP download. Please try again.");
    }
  };

  return (
    <div className="space-y-5">
      <FileUploader
        accept="application/pdf,.pdf"
        files={file ? [file] : []}
        onFilesAdded={handleFilesAdded}
        onFileRemove={handleRemove}
        hint="Upload a single PDF file to split."
        disabled={isProcessing}
      />

      {pageCount > 0 && (
        <p className="text-xs text-muted-foreground">
          This PDF has <span className="font-medium">{pageCount}</span> pages.
        </p>
      )}

      <div className="space-y-2">
        <Label htmlFor="ranges">Page ranges</Label>
        <Input
          id="ranges"
          value={rangesInput}
          onChange={(e) => setRangesInput(e.target.value)}
          placeholder="e.g. 1-3, 5, 7-9"
          disabled={isProcessing}
        />
        <p className="text-xs text-muted-foreground">
          Each range produces a separate PDF. Use commas to separate multiple ranges.
        </p>
      </div>

      {error && (
        <p className="rounded-md bg-destructive/10 p-3 text-sm text-destructive" role="alert">
          {error}
        </p>
      )}

      {isProcessing && (
        <div className="space-y-2">
          <div className="flex items-center justify-between gap-3">
            <p className="text-xs text-muted-foreground">
              {cancellation.isCancelling
                ? "Stopping after the current page range..."
                : `Splitting PDF... ${progress}%`}
            </p>
            <CancelOperationButton
              onCancel={cancellation.cancel}
              isCancelling={cancellation.isCancelling}
              cancelLabel="Cancel after this range"
              cancellingLabel="Stopping after this range..."
            />
          </div>
          <Progress value={progress} />
        </div>
      )}

      <Button
        onClick={handleSplit}
        disabled={!file || isProcessing}
        className="w-full"
        size="lg"
      >
        {isProcessing ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null}
        {isProcessing ? "Splitting..." : "Split PDF"}
      </Button>

      {results.length > 0 && (
        <div className="space-y-3">
          <div className="flex flex-wrap items-center justify-between gap-2">
            <h3 className="text-sm font-semibold">
              Results ({results.length} file{results.length === 1 ? "" : "s"})
            </h3>
            <div className="flex flex-wrap gap-2">
              <Button
                size="sm"
                variant="outline"
                onClick={handleProcessAnotherFile}
              >
                Process another file
              </Button>
              {results.length > 1 && (
                <Button
                  size="sm"
                  variant="outline"
                  onClick={handleDownloadAllZip}
                >
                  <Archive className="mr-1 h-3.5 w-3.5" /> Download all as ZIP
                </Button>
              )}
            </div>
          </div>
          <ul className="space-y-2">
            {results.map((r, i) => (
              <li
                key={i}
                className="flex items-center justify-between rounded-lg border bg-card p-3 text-sm"
              >
                <div className="min-w-0">
                  <p className="truncate font-medium">{r.name}</p>
                  <p className="text-xs text-muted-foreground">
                    {formatBytes(r.bytes.byteLength)}
                  </p>
                </div>
                <Button
                  size="sm"
                  variant="outline"
                  onClick={() =>
                    downloadBytes(r.bytes, r.name, "application/pdf")
                  }
                >
                  <Download className="mr-1 h-3.5 w-3.5" /> Download
                </Button>
              </li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}

"use client";

import { useCallback, useEffect, useRef, useState } from "react";

export function useOperationCancellation() {
  const controllerRef = useRef<AbortController | null>(null);
  const [isCancelling, setIsCancelling] = useState(false);

  const start = useCallback(() => {
    const controller = new AbortController();
    controllerRef.current = controller;
    setIsCancelling(false);
    return controller;
  }, []);

  const cancel = useCallback(() => {
    controllerRef.current?.abort();
    setIsCancelling(true);
  }, []);

  const finish = useCallback((controller: AbortController) => {
    if (controllerRef.current === controller) {
      controllerRef.current = null;
      setIsCancelling(false);
    }
  }, []);

  useEffect(
    () => () => {
      controllerRef.current?.abort();
    },
    []
  );

  return { start, cancel, finish, isCancelling };
}
