/**
 * Pre-render every tool page at build time (SSG).
 * This makes each tool page fast and SEO-friendly.
 */
export function generateStaticParams() {
  return [...TOOLS.map((t) => ({ slug: t.slug })), { slug: "pdf-to-text" }];
}

/**
 * Generate unique SEO metadata for each tool page.
 * Uses the centralized buildToolMetadata helper which produces:
 *  - unique title and concise description
 *  - canonical: https://convaudit.com/tools/{slug}
 *  - openGraph: title, description, url, type, siteName
 *  - twitter: summary_large_image card
 *  - alternates: canonical + languages (en, ar) for hreflang
 *  - robots: index, follow
 *
 * OG image is auto-generated by app/tools/[slug]/opengraph-image.tsx.
 */
export function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string }>;
}): Promise<Metadata> {
  return (async () => {
    const { slug } = await params;
    const tool = getToolBySlug(slug);
    if (!tool) {
      return { title: "Tool not found" };
    }
    return buildToolMetadata(tool);
  })();
}

/**
 * Tool page â€” renders the ToolLayout with the matching tool component.
 */
export default async function ToolPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  const tool = getToolBySlug(slug);
  if (!tool) notFound();

  if (slug === "compress-image") {
    return <CompressImageCitationPage tool={tool} />;
  }

  return (
    <ToolLayout tool={tool}>
      <ToolRenderer slug={slug} />
    </ToolLayout>
  );
}

function sanitizeAdsenseClient(value: string | undefined): string {

function matchesAcceptedType(file: File, accept: string): boolean {
  const acceptedTypes = accept
    .split(",")
    .map((type) => type.trim().toLowerCase())
    .filter(Boolean);

  return acceptedTypes.some((type) => {
    if (type.startsWith(".")) {
      return file.name.toLowerCase().endsWith(type);
    }
    if (type.endsWith("/*")) {
      return file.type.toLowerCase().startsWith(type.slice(0, -1));
    }
    return file.type.toLowerCase() === type;
  });
}

export function validateToolFile(file: File, accept: string): string | null {
  if (!matchesAcceptedType(file, accept)) {
    return `${file.name} is not a supported file type. Choose a supported file.`;
  }
  return null;
}

"use client";

import { saveAs } from "file-saver";

/**
 * Trigger a browser download for a single Blob using file-saver.
 */
export function downloadBlob(blob: Blob, filename: string) {
  saveAs(blob, filename);
}

/**
 * Trigger a download for raw bytes by wrapping them in a Blob.
 */
export function downloadBytes(bytes: Uint8Array, filename: string, mime: string) {
  // Copy into a fresh ArrayBuffer so file-saver's typing is happy.
  const ab = new ArrayBuffer(bytes.byteLength);
  new Uint8Array(ab).set(bytes);
  downloadBlob(new Blob([ab], { type: mime }), filename);
}

/**
 * Copy text to the clipboard.
 *
 * navigator.clipboard.writeText() only exists in secure contexts
 * (HTTPS or localhost) and can reject when permission is denied.
 * Every copy button in the app called it bare â€” on plain HTTP
 * (or with clipboard permission blocked) the click handler threw
 * "Cannot read properties of undefined" and the "Copied!" UI
 * showed even though nothing was copied. This helper falls back
 * to the legacy execCommand path and returns success/failure.
 */
export async function copyText(text: string): Promise<boolean> {
  try {
    if (
      typeof navigator !== "undefined" &&
      navigator.clipboard &&
      typeof navigator.clipboard.writeText === "function"
    ) {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch {
    // fall through to legacy path
  }
  try {
    const ta = document.createElement("textarea");
    ta.value = text;
    ta.style.cssText = "position:fixed;opacity:0;pointer-events:none;";
    document.body.appendChild(ta);
    ta.select();
    const ok = document.execCommand("copy");
    ta.remove();
    return ok;
  } catch {
    return false;
  }
}

/**
 * Generate a short unique id (good enough for client-side lists).
 */
export function shortId(): string {
  return Math.random().toString(36).slice(2, 10);
}

/**
 * Format bytes into a human-readable string.
 */
export function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
}

/**
 * Extract the 11-character video ID from any common YouTube URL format:
 *  - https://www.youtube.com/watch?v=VIDEO_ID
 *  - https://youtu.be/VIDEO_ID
 *  - https://www.youtube.com/embed/VIDEO_ID
 *  - https://www.youtube.com/shorts/VIDEO_ID
 *  - https://www.youtube.com/live/VIDEO_ID
 * Returns null if no valid ID is found.
 */
export function extractVideoId(input: string): string | null {
  const trimmed = input.trim();
  if (!trimmed) return null;

  if (/^[a-zA-Z0-9_-]{11}$/.test(trimmed)) return trimmed;

  try {
    const url = new URL(trimmed);
    if (
      url.protocol !== "https:" ||
      url.username ||
      url.password ||
      url.port ||
      !["youtube.com", "www.youtube.com", "m.youtube.com", "youtu.be"].includes(
        url.hostname.toLowerCase(),
      )
    ) {
      return null;
    }

    const match =
      url.hostname.toLowerCase() === "youtu.be"
        ? url.pathname.match(/^\/([A-Za-z0-9_-]{11})\/?$/)
        : url.pathname === "/watch"
          ? [null, url.searchParams.get("v") ?? ""]
          : url.pathname.match(/^\/(?:embed|shorts|live)\/([A-Za-z0-9_-]{11})\/?$/);
    const id = match?.[1] ?? null;
    return id && /^[A-Za-z0-9_-]{11}$/.test(id) ? id : null;
  } catch {
    return null;
  }
}

/**
 * Build all standard thumbnail URLs for a video ID.
 * maxresdefault may return a 404 for older videos â€” callers should handle
 * image load errors and fall back to sd/hq.
 */
export function getThumbnailUrls(videoId: string): ThumbnailSet {
  const base = `https://img.youtube.com/vi/${videoId}`;
  return {
    maxres: `${base}/maxresdefault.jpg`,
    sd: `${base}/sddefault.jpg`,
    hq: `${base}/hqdefault.jpg`,
    mq: `${base}/mqdefault.jpg`,
    default: `${base}/default.jpg`,
  };
}

/**
 * Fetch oEmbed metadata for a YouTube video URL.
 * The oEmbed endpoint supports CORS and requires no API key.
 * Throws on network error or non-OK response.
 */
export async function fetchOEmbed(
  videoUrl: string,
  signal?: AbortSignal
): Promise<YouTubeOEmbed> {
  const endpoint = `https://www.youtube.com/oembed?url=${encodeURIComponent(
    videoUrl
  )}&format=json`;
  const res = await fetch(endpoint, { signal });
  if (!res.ok) {
    throw new Error(
      res.status === 404
        ? "Video not found. Check the URL and try again."
        : `Failed to fetch video metadata (HTTP ${res.status}).`
    );
  }
  return (await res.json()) as YouTubeOEmbed;
}

/**
 * Parse a timestamp string ("1:23", "01:23:45", "83") into seconds.
 * Returns -1 for invalid input so callers can distinguish "0:00"
 * (valid) from garbage (invalid). Previously every invalid string
 * returned 0, so "abc" silently generated a link to the start of
 * the video.
 */
export function parseTimestampToSeconds(input: string): number {
  const trimmed = input.trim();
  if (!trimmed) return -1;

  // Pure number â†’ seconds
  if (/^\d+$/.test(trimmed)) return parseInt(trimmed, 10);

  const parts = trimmed.split(":").map((p) => p.trim());
  if (parts.some((p) => !/^\d+$/.test(p))) return -1;

  if (parts.length === 2) {
    // MM:SS
    return parseInt(parts[0], 10) * 60 + parseInt(parts[1], 10);
  }
  if (parts.length === 3) {
    // HH:MM:SS
    return (
      parseInt(parts[0], 10) * 3600 +
      parseInt(parts[1], 10) * 60 +
      parseInt(parts[2], 10)
    );
  }
  return -1;
}

/**
 * Format seconds as a human-readable timestamp.
 * 83 â†’ "1:23", 3661 â†’ "1:01:01"
 */
export function formatSecondsToTimestamp(seconds: number): string {
  const h = Math.floor(seconds / 3600);
  const m = Math.floor((seconds % 3600) / 60);
  const s = seconds % 60;
  if (h > 0) {
    return `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
  }
  return `${m}:${String(s).padStart(2, "0")}`;
}

/**
 * Build a shareable YouTube timestamp link.
 * https://youtu.be/VIDEO_ID?t=SECONDS
 */
export function buildTimestampLink(videoId: string, seconds: number): string {
  return `https://youtu.be/${videoId}?t=${seconds}`;
}

/**
 * Extract a channel ID (UC...) from a YouTube channel URL.
 * Returns null if the URL doesn't contain a channel ID.
 */
export function extractChannelId(url: string): string | null {
  const match = url.match(/\/channel\/(UC[a-zA-Z0-9_-]{22})/);
  return match ? match[1] : null;
}

/**
 * Format a USD amount for display.
 */
export function formatCurrency(amount: number): string {
  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    maximumFractionDigits: 0,
  }).format(amount);
}

/**
 * Format a large number (e.g. 1.5M views).
 */
export function formatLargeNumber(n: number): string {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
  return n.toString();
}

  Object.keys(CPM_RANGES) as YouTubeNiche[]
).map((niche) => ({
  niche,
  cpmLow: CPM_RANGES[niche].low,
  cpmHigh: CPM_RANGES[niche].high,

export type EarningsEstimate = {
  low: number;
  mid: number;
  high: number;
};

export type YouTubeEarnings = {
  monthly: EarningsEstimate;
  annual: EarningsEstimate;
  rpm: EarningsEstimate;
};

function roundToCents(amount: number): number {
  return Math.round((amount + Number.EPSILON) * 100) / 100;
}

export function calculateEarnings(
  views: number,
  niche: NicheCPM,
): YouTubeEarnings {
  const cpmMid = (niche.cpmLow + niche.cpmHigh) / 2;
  const rpm = {
    low: roundToCents(niche.cpmLow * CREATOR_SHARE),
    mid: roundToCents(cpmMid * CREATOR_SHARE),
    high: roundToCents(niche.cpmHigh * CREATOR_SHARE),
  };
  const monthly = {
    low: roundToCents((views / 1000) * rpm.low),
    mid: roundToCents((views / 1000) * rpm.mid),
    high: roundToCents((views / 1000) * rpm.high),
  };

  return {
    monthly,
    rpm,
    annual: {
      low: roundToCents(monthly.low * 12),
      mid: roundToCents(monthly.mid * 12),
      high: roundToCents(monthly.high * 12),
    },
  };
}
