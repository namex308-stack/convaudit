"use client";

import { useState } from "react";
import { Loader2, Search, AlertCircle, Copy, Check, User, Youtube } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import {
  extractVideoId,
  fetchOEmbed,
  extractChannelId,
  type YouTubeOEmbed,
} from "@/lib/youtube";
import { copyText } from "@/lib/download";
import { CancelOperationButton } from "../CancelOperationButton";
import { useOperationCancellation } from "../useOperationCancellation";

type Result = {
  channelId: string;
  channelName: string;
  channelUrl: string;
};

/**
 * YouTube Channel ID Finder.
 *
 * Strategy:
 * 1. If the input is a video URL, use oEmbed to get author_url (which
 *    contains the channel ID). This is the most reliable method.
 * 2. If the input is a channel URL (/channel/UC...), extract directly.
 * 3. If the input is a handle URL (/@name), use a CORS proxy to fetch
 *    the channel page and extract the ID from meta tags.
 */
export function YouTubeChannelIdFinder() {
  const [url, setUrl] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [result, setResult] = useState<Result | null>(null);
  const [meta, setMeta] = useState<YouTubeOEmbed | null>(null);
  const [copied, setCopied] = useState(false);
  const cancellation = useOperationCancellation();

  const handleFind = async () => {
    setError(null);
    setResult(null);
    setMeta(null);
    setCopied(false);

    const trimmed = url.trim();
    if (!trimmed) {
      setError("Please enter a YouTube URL.");
      return;
    }

    setLoading(true);
    const controller = cancellation.start();
    try {
      // Strategy 1: Video URL â†’ oEmbed â†’ author_url
      const videoId = extractVideoId(trimmed);
      if (videoId) {
        const oembed = await fetchOEmbed(
          `https://www.youtube.com/watch?v=${videoId}`,
          controller.signal
        );
        const channelId = extractChannelId(oembed.author_url);
        if (channelId) {
          setMeta(oembed);
          setResult({
            channelId,
            channelName: oembed.author_name,
            channelUrl: oembed.author_url,
          });
          return;
        }
      }

      // Strategy 2: Direct channel URL (/channel/UC...)
      const directId = extractChannelId(trimmed);
      if (directId) {
        setResult({
          channelId: directId,
          channelName: "Channel",
          channelUrl: trimmed,
        });
        return;
      }

      // Strategy 3: Handle URL (/@name) â†’ fetch page via CORS proxy
      if (trimmed.includes("/@") || trimmed.includes("/user/") || trimmed.includes("/c/")) {
        const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(trimmed)}`;
        const res = await fetch(proxyUrl, { signal: controller.signal });
        if (res.ok) {
          const html = await res.text();
          // Look for <meta itemprop="channelId" content="UC...">
          const idMatch = html.match(/"channelId":"(UC[a-zA-Z0-9_-]{22})"/);
          if (idMatch) {
            // Also try to extract channel name
            const nameMatch = html.match(/<meta\s+property="og:title"\s+content="([^"]*)"/);
            setResult({
              channelId: idMatch[1],
              channelName: nameMatch ? nameMatch[1] : "Channel",
              channelUrl: trimmed,
            });
            return;
          }
        }
        setError("Could not resolve channel ID from this URL. Try a video URL from the channel instead.");
        return;
      }

      setError("Please enter a valid YouTube URL â€” a video URL, channel URL, or @handle URL.");
    } catch (err) {
      setError(controller.signal.aborted
        ? "Lookup cancelled."
        : err instanceof Error
          ? err.message
          : "Failed to find channel ID. Please try again.");
    } finally {
      cancellation.finish(controller);
      setLoading(false);
    }
  };

  const copyId = async () => {
    if (result) {
      const ok = await copyText(result.channelId);
      setCopied(ok);
      if (ok) setTimeout(() => setCopied(false), 2000);
    }
  };

  return (
    <div className="space-y-5">
      <div className="space-y-2">
        <Label htmlFor="yt-url">YouTube URL (video, channel, or @handle)</Label>
        <div className="flex flex-col gap-2 sm:flex-row">
          <Input
            id="yt-url"
            type="url"
            value={url}
            onChange={(e) => setUrl(e.target.value)}
            placeholder="https://www.youtube.com/@MrBeast or /watch?v=..."
            onKeyDown={(e) => e.key === "Enter" && handleFind()}
            disabled={loading}
          />
          <Button onClick={handleFind} disabled={loading || !url.trim()} className="sm:w-auto">
            {loading ? <Loader2 className="mr-1.5 h-4 w-4 animate-spin" /> : <Search className="mr-1.5 h-4 w-4" />}
            Find Channel ID
          </Button>
          {loading && (
            <CancelOperationButton
              onCancel={cancellation.cancel}
              isCancelling={cancellation.isCancelling}
            />
          )}
        </div>
        <p className="text-xs text-muted-foreground">
          Video URLs are the most reliable. Handle URLs (@name) use a CORS proxy and may occasionally fail.
        </p>
      </div>

      {error && (
        <div className="flex items-start gap-2 rounded-md bg-destructive/10 p-3 text-sm text-destructive" role="alert">
          <AlertCircle className="mt-0.5 h-4 w-4 flex-none" />
          <span>{error}</span>
        </div>
      )}

      {result && (
        <div className="space-y-4">
          {meta && (
            <div className="flex items-center gap-3 rounded-lg border bg-muted/30 p-3">
              <Youtube className="h-5 w-5 flex-none text-red-600" />
              <div className="min-w-0">
                <p className="truncate text-sm font-medium">{meta.title}</p>
                <p className="truncate text-xs text-muted-foreground">Video used for lookup</p>
              </div>
            </div>
          )}

          <div className="rounded-xl border bg-card p-5">
            <div className="mb-3 flex items-center gap-2 text-xs font-medium uppercase tracking-wider text-muted-foreground">
              <User className="h-3.5 w-3.5" />
              Channel ID
            </div>
            <div className="flex items-center justify-between gap-3">
              <code className="break-all rounded bg-muted px-3 py-2 font-mono text-sm font-medium">
                {result.channelId}
              </code>
              <Button size="sm" variant="outline" className="flex-none" onClick={copyId}>
                {copied ? <Check className="mr-1 h-3.5 w-3.5 text-emerald-600" /> : <Copy className="mr-1 h-3.5 w-3.5" />}
                {copied ? "Copied!" : "Copy"}
              </Button>
            </div>
            <div className="mt-3 space-y-1 text-sm text-muted-foreground">
              <p><strong className="text-foreground">Channel name:</strong> {result.channelName}</p>
              <p><strong className="text-foreground">Channel URL:</strong>{" "}
                <a href={result.channelUrl} target="_blank" rel="noopener noreferrer" className="text-foreground underline underline-offset-2 hover:text-muted-foreground">
                  {result.channelUrl}
                </a>
              </p>
            </div>
          </div>

          <div className="rounded-lg border border-dashed border-border/60 bg-muted/20 p-3 text-xs text-muted-foreground">
            Use this ID for the YouTube Data API, RSS feeds ({`https://www.youtube.com/feeds/videos.xml?channel_id=${result.channelId}`}), or third-party analytics tools.
          </div>
        </div>
      )}
    </div>
  );
}

/**
 * YouTube Description Generator.
 *
 * Builds a structured description from user input: hook, summary,
 * timestamp chapters, social links, and hashtags. Output is editable
 * and copyable.
 */
export function YouTubeDescriptionGenerator() {
  const [title, setTitle] = useState("");
  const [summary, setSummary] = useState("");
  const [timestamps, setTimestamps] = useState<Timestamp[]>([
    { id: "ts-1", time: "0:00", label: "Intro" },
    { id: "ts-2", time: "", label: "" },
  ]);
  const [socialLinks, setSocialLinks] = useState({
    twitter: "",
    instagram: "",
    website: "",
  });
  const [hashtags, setHashtags] = useState("");
  const [copied, setCopied] = useState(false);

  const addTimestamp = () => {
    setTimestamps((prev) => [
      ...prev,
      { id: `ts-${Date.now()}`, time: "", label: "" },
    ]);
  };

  const removeTimestamp = (id: string) => {
    setTimestamps((prev) => prev.filter((t) => t.id !== id));
  };

  const updateTimestamp = (id: string, field: "time" | "label", value: string) => {
    setTimestamps((prev) =>
      prev.map((t) => (t.id === id ? { ...t, [field]: value } : t))
    );
  };

  const generateDescription = (): string => {
    const lines: string[] = [];

    // Hook (first 1-2 lines â€” most visible in search)
    if (summary.trim()) {
      lines.push(summary.trim());
      lines.push("");
    }

    // Timestamps
    const validTs = timestamps.filter((t) => t.time.trim() && t.label.trim());
    if (validTs.length > 0) {
      lines.push("â±ï¸ Timestamps:");
      validTs.forEach((t) => {
        lines.push(`${t.time.trim()} ${t.label.trim()}`);
      });
      lines.push("");
    }

    // About section
    if (title.trim()) {
      lines.push(`About this video:`);
      lines.push(`In this video, we cover ${title.trim()}. Whether you're a beginner or looking to level up, this guide breaks down everything you need to know.`);
      lines.push("");
    }

    // Social links
    const hasSocial = Object.values(socialLinks).some((v) => v.trim());
    if (hasSocial) {
      lines.push("ðŸ”— Connect with me:");
      if (socialLinks.twitter.trim()) lines.push(`Twitter: ${socialLinks.twitter.trim()}`);
      if (socialLinks.instagram.trim()) lines.push(`Instagram: ${socialLinks.instagram.trim()}`);
      if (socialLinks.website.trim()) lines.push(`Website: ${socialLinks.website.trim()}`);
      lines.push("");
    }

    // Hashtags
    if (hashtags.trim()) {
      lines.push(hashtags.trim());
      lines.push("");
    }

    // Disclaimer
    lines.push("---");
    lines.push("If you found this video helpful, please like and subscribe! ðŸ””");
    lines.push("");
    lines.push(`# ${title.trim() || "Your Video Title"}`);

    return lines.join("\n");
  };

  const handleCopy = async () => {
    const desc = generateDescription();
    const ok = await copyText(desc);
    setCopied(ok);
    if (ok) setTimeout(() => setCopied(false), 2000);
  };

  const description = generateDescription();

  return (
    <div className="space-y-5">
      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
        <div className="space-y-2">
          <Label htmlFor="title">Video title</Label>
          <Input
            id="title"
            value={title}
            onChange={(e) => setTitle(e.target.value)}
            placeholder="e.g. How to Learn Guitar Fast"
          />
        </div>
        <div className="space-y-2">
          <Label htmlFor="summary">Hook / opening line</Label>
          <Input
            id="summary"
            value={summary}
            onChange={(e) => setSummary(e.target.value)}
            placeholder="One sentence that makes viewers want to watch"
          />
        </div>
      </div>

      {/* Timestamps */}
      <div className="space-y-2">
        <Label>Timestamps (for video chapters)</Label>
        <div className="space-y-2">
          {timestamps.map((ts) => (
            <div key={ts.id} className="flex items-center gap-2">

                onChange={(e) => updateTimestamp(ts.id, "time", e.target.value)}

              />
              <Input
                value={ts.label}
                onChange={(e) => updateTimestamp(ts.id, "label", e.target.value)}

              />
              <Button

                disabled={timestamps.length <= 1}
              >
                <Trash2 className="h-4 w-4" />
              </Button>
            </div>
          ))}

          <Plus className="mr-1 h-3.5 w-3.5" /> Add timestamp

"use client";

import { useState } from "react";
import { Hash, Copy, Check, Sparkles } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Badge } from "@/components/ui/badge";
import { copyText } from "@/lib/download";

/**
 * Generate hashtags from a topic by combining it with proven patterns.
 * Returns a deduplicated list.
 */
function generateHashtags(topic: string): string[] {
  const t = topic.trim().toLowerCase().replace(/\s+/g, "");
  if (!t) return [];
  const year = new Date().getFullYear();

  const topicWords = topic.trim().toLowerCase().split(/\s+/).filter(Boolean);
  const topicPhrase = topicWords.join("");

  const tags = new Set<string>();

  // Core topic tags
  tags.add(`#${topicPhrase}`);
  topicWords.forEach((w) => tags.add(`#${w}`));

  // Pattern-based variations
  const patterns = [
    `#${topicPhrase}tips`,
    `#${topicPhrase}tutorial`,
    `#${topicPhrase}guide`,
    `#${topicPhrase}${year}`,
    `#${topicPhrase}hacks`,
    `#${topicPhrase}forbeginners`,
    `#${topicPhrase}masterclass`,
    `#${topicPhrase}explained`,
    `#howto${topicPhrase}`,
    `#learn${topicPhrase}`,
    `#${topicPhrase}community`,
    `#${topicPhrase}love`,
    `#${topicPhrase}daily`,
    `#${topicPhrase}tipsandtricks`,
  ];
  patterns.forEach((p) => tags.add(p));

  // General content tags
  const general = [
    "#youtube",
    "#tutorial",
    "#howto",
    "#viral",
    "#trending",
    `#${year}`,
    "#shorts",
    "#educational",
    "#learning",
    "#tips",
  ];
  general.forEach((g) => tags.add(g));

  return Array.from(tags);
}

/**
 * YouTube Hashtag Generator.
 *
 * Generates topic-specific + general hashtags. Click to copy individual
 * tags, or copy all. Reminds users to use 5-10 tags max.
 */
export function YouTubeHashtagGenerator() {
  const [topic, setTopic] = useState("");
  const [results, setResults] = useState<string[]>([]);
  const [copiedAll, setCopiedAll] = useState(false);
  const [copiedTag, setCopiedTag] = useState<string | null>(null);

  const handleGenerate = () => {
    setResults(generateHashtags(topic));
  };

  const copyAll = async () => {
    const ok = await copyText(results.slice(0, 10).join(" "));
    setCopiedAll(ok);
    if (ok) setTimeout(() => setCopiedAll(false), 2000);
  };

  const copyTag = async (tag: string) => {
    const ok = await copyText(tag);
    if (ok) {
      setCopiedTag(tag);
      setTimeout(() => setCopiedTag(null), 2000);
    }
  };

  return (
    <div className="space-y-5">
      <div className="space-y-2">
        <Label htmlFor="topic">Your video topic or niche</Label>
        <div className="flex flex-col gap-2 sm:flex-row">
          <Input
            id="topic"
            value={topic}
            onChange={(e) => setTopic(e.target.value)}
            placeholder="e.g. personal finance, vegan cooking, guitar..."
            onKeyDown={(e) => e.key === "Enter" && handleGenerate()}
          />
          <Button onClick={handleGenerate} disabled={!topic.trim()} className="sm:w-auto">
            <Sparkles className="mr-1.5 h-4 w-4" />
            Generate Hashtags
          </Button>
        </div>
        <p className="text-xs text-muted-foreground">
          We recommend using 5-10 hashtags in your description. The first 3 appear above your title on mobile.
        </p>
      </div>

      {results.length > 0 && (
        <div className="space-y-4">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2">
              <Hash className="h-4 w-4 text-muted-foreground" />
              <span className="text-sm font-medium">
                {results.length} hashtags generated
              </span>
            </div>
            <Button size="sm" variant="outline" onClick={copyAll}>
              {copiedAll ? <Check className="mr-1 h-3.5 w-3.5 text-emerald-600" /> : <Copy className="mr-1 h-3.5 w-3.5" />}
              {copiedAll ? "Copied!" : "Copy top 10"}
            </Button>
          </div>

          <div className="flex flex-wrap gap-2">
            {results.map((tag, idx) => (
              <button
                key={tag}
                onClick={() => copyTag(tag)}
                title="Click to copy"
                className="group"
              >
                <Badge
                  variant={idx < 3 ? "default" : "secondary"}
                  className={cn(
                    "cursor-pointer text-xs transition-all hover:scale-105",
                    idx < 3 && "font-medium"
                  )}
                >
                  {copiedTag === tag && <Check className="mr-1 h-3 w-3" />}
                  {tag}
                </Badge>
              </button>
            ))}
          </div>

          <div className="rounded-lg border border-dashed border-border/60 bg-muted/20 p-3 text-xs text-muted-foreground">
            <strong className="text-foreground">Tip:</strong> The first 3 hashtags (highlighted) will appear above your title on mobile. Choose the most relevant ones for those slots.
          </div>
        </div>
      )}
    </div>
  );
}

          method: "POST",
          credentials: "omit",
          cache: "no-store",

          signal: controller.signal,
        });

  throwIfCancelled,
  validateToolFile,
} from "@/lib/file-validation";

const TARGET_W = 1280;
const TARGET_H = 720;

/**
 * YouTube Thumbnail Resizer.
 *
 * Loads an image into a canvas viewport at 1280Ã—720, lets the user drag
 * to reposition, optionally shows a safe-zone overlay (central 80% where
 * text/faces won't be clipped by YouTube's mobile UI), and exports the
 * cropped result as a PNG.
 */
export function YouTubeThumbnailResizer() {
  const [imageData, setImageData] = useState<{
    url: string;
    img: HTMLImageElement;
    width: number;
    height: number;
  } | null>(null);
  const [offset, setOffset] = useState({ x: 0, y: 0 });
  const [showSafeZone, setShowSafeZone] = useState(true);
  const [isExporting, setIsExporting] = useState(false);
  const [wasExported, setWasExported] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const cancellation = useOperationCancellation();
  const dragRef = useRef<{ startX: number; startY: number; baseX: number; baseY: number } | null>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const handleFile = useCallback((file: File) => {
    setError(null);
    const validationError = validateToolFile(
      file,
      "image/*,.heic,.heif,.tiff,.tif,.bmp,.gif,.avif"
    );
    if (validationError) {
      setError(validationError);
      return;
    }
    setWasExported(false);
    // Revoke the previous object URL to avoid leaking memory on
    // repeated uploads.
    setImageData((prev) => {
      if (prev) URL.revokeObjectURL(prev.url);
      return prev;
    });
    const url = URL.createObjectURL(file);
    const img = new Image();
    img.onload = () => {
      setImageData({ url, img, width: img.naturalWidth, height: img.naturalHeight });
      setOffset({ x: 0, y: 0 });
    };
    img.onerror = () => {
      URL.revokeObjectURL(url);
      setError("Failed to load image. Try a different file.");
    };
    img.src = url;
  }, []);

  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    const file = e.dataTransfer.files[0];
    if (file) handleFile(file);
  };

  /**
   * Calculate the draw dimensions for "cover" fit: the image fills the
   * 1280Ã—720 frame, overflowing if necessary. The user drags to choose
   * which part is visible.
   */
  const getDrawDimensions = () => {
    if (!imageData) return { drawW: TARGET_W, drawH: TARGET_H, scale: 1 };
    const scale = Math.max(TARGET_W / imageData.width, TARGET_H / imageData.height);
    return {
      drawW: imageData.width * scale,
      drawH: imageData.height * scale,
      scale,
    };
  };

  const handleMouseDown = (e: React.MouseEvent) => {
    if (!imageData) return;
    dragRef.current = {
      startX: e.clientX,
      startY: e.clientY,
      baseX: offset.x,
      baseY: offset.y,
    };
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    if (!dragRef.current || !imageData) return;
    const dx = e.clientX - dragRef.current.startX;
    const dy = e.clientY - dragRef.current.startY;
    const { drawW, drawH } = getDrawDimensions();
    // Clamp so the image always covers the frame
    const maxX = Math.max(0, (drawW - TARGET_W) / 2);
    const maxY = Math.max(0, (drawH - TARGET_H) / 2);
    setOffset({
      x: Math.max(-maxX, Math.min(maxX, dragRef.current.baseX + dx)),
      y: Math.max(-maxY, Math.min(maxY, dragRef.current.baseY + dy)),
    });
  };

      // Draw the image centered + offset, covering the frame
      ctx.drawImage(
        imageData.img,
        (TARGET_W - drawW) / 2 + offset.x,
        (TARGET_H - drawH) / 2 + offset.y,
        drawW,
        drawH
      );

      const blob = await new Promise<Blob>((resolve, reject) =>

          (b) => (b ? resolve(b) : reject(new Error("Failed to export"))),
          "image/png"

      );
      throwIfCancelled(controller.signal);
      downloadBlob(blob, `youtube-thumbnail-${shortId()}.png`);
      setWasExported(true);
    } catch (err) {
      if (controller.signal.aborted) {
        setError("Export cancelled. The image was not downloaded.");
      } else {
        console.error(err);
        setError("Failed to export thumbnail. Please try again.");
      }
    } finally {
      cancellation.finish(controller);
      setIsExporting(false);
    }
  };

  const { drawW, drawH } = getDrawDimensions();
  const processAnotherImage = () => {
    if (imageData) URL.revokeObjectURL(imageData.url);
    setImageData(null);
    setOffset({ x: 0, y: 0 });
    setShowSafeZone(true);
    setWasExported(false);
    setError(null);
  };

  return (
    <div className="space-y-5">
      <input
        ref={fileInputRef}
        type="file"
        accept="image/*,.heic,.heif,.tiff,.tif,.bmp,.gif,.avif"
        className="hidden"
        onChange={(e) => {
          const f = e.target.files?.[0];
          if (f) handleFile(f);
          e.currentTarget.value = "";
        }}

        <div
          onDrop={handleDrop}
          onDragOver={(e) => e.preventDefault()}
          onClick={() => fileInputRef.current?.click()}
          role="button"
          tabIndex={0}
          aria-label={`Upload a thumbnail image. ${FILE_SIZE_GUIDANCE}`}
          onKeyDown={(e) => {
            if (e.key === "Enter" || e.key === " ") {
              e.preventDefault();
              fileInputRef.current?.click();
            }
          }}
          className="flex min-h-50 cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed border-border p-8 text-center transition-colors hover:border-primary/50 hover:bg-muted/30 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
        >
          <Upload className="h-10 w-10 text-muted-foreground" />
          <p className="mt-3 text-sm font-medium">Drop an image here, or click to browse</p>
          <p className="mt-1 text-xs text-muted-foreground">
            JPG, PNG, WebP, and more Â· {FILE_SIZE_GUIDANCE}

      ) : (
        <>

              <p className="text-xs text-muted-foreground">Drag to reposition â€¢ Output: 1280Ã—720</p>

type FieldKey = (typeof fields)[number]["key"];
type Counts = Record<FieldKey, string>;

function countValue(value: string) {
  const parsed = Number(value);
  return Number.isFinite(parsed) ? Math.max(0, parsed) : 0;
}

    countValue(counts.likes) +

    countValue(counts.shares);
